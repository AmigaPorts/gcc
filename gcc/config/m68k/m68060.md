;; The 68060 has two Operand Execution Pipelines (OEPs):
;;   pOEP (primary)   -- executes all instructions
;;   sOEP (secondary) -- executes a subset of simple instructions
;;
;; Up to two instructions per cycle when pairing rules are satisfied.
;; Six dispatch tests must all pass for dual-issue:
;;   Test 1: sOEP instruction words in buffer (not modeled -- fetch effect)
;;   Test 2: Instruction classification compatibility
;;   Test 3: sOEP EA mode restrictions (no full format (bd,An,Xn), no PC-rel)
;;   Test 4: At most one data memory reference per pair
;;   Test 5: No sOEP.AGU register conflict with pOEP (handled by scheduler)
;;   Test 6: No sOEP.IEE register conflict with pOEP (handled by scheduler)

;; =========================================================================
;; Automata and CPU units
;; =========================================================================

(define_automaton "m68060_oep")
(define_cpu_unit "m68060_pOEP, m68060_sOEP" "m68060_oep")

;; sOEP can only dispatch if pOEP also dispatches (pair with primary).
(final_presence_set "m68060_sOEP" "m68060_pOEP")

;; Single-issue reservation: occupies both pipelines.
(define_reservation "m68060_single" "m68060_pOEP + m68060_sOEP")

;; Memory unit: Test 4 -- at most one data memory access per pair.
(define_automaton "m68060_mem")
(define_cpu_unit "m68060_mem" "m68060_mem")

;; =========================================================================
;; Instruction reservations
;; =========================================================================

;; The reservations by type are for patterns that emit one instruction;
;; sched_insns (m68k.md) counts the instructions of the others, which have
;; their own reservations at the end.

;; --- Single-cycle integer instructions ------------------------------------
;; m68060_pair (m68k.md) says which pipelines take each one.  The first of
;; a pair takes the pOEP, and the second the sOEP, which the
;; final_presence_set above allows only next to a pOEP instruction.  A
;; memory access also takes the memory unit (Test 4).

;; pOEP|sOEP: no memory access.
(define_insn_reservation "m68060_both_00" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
				  shift,moveq_l,mov3q_l,move_l,mvsz,
				  tst,clr,cmp,move")
		 (and (eq_attr "op_mem" "00")
		      (eq_attr "m68060_pair" "both")))))
  "m68060_pOEP | m68060_sOEP")

;; pOEP|sOEP: a load or a store, indexed or not.
(define_insn_reservation "m68060_both_mem" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
				  shift,moveq_l,mov3q_l,move_l,mvsz,
				  tst,clr,cmp,move")
		 (and (eq_attr "op_mem" "10,i0,01,0i")
		      (eq_attr "m68060_pair" "both")))))
  "(m68060_pOEP + m68060_mem) | (m68060_sOEP + m68060_mem)")

;; pOEP, with an sOEP instruction beside it: an address the sOEP doesn't
;; take (Test 3), or an immediate with a memory operand (pOEP-until-last).
(define_insn_reservation "m68060_first_00" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
				  shift,moveq_l,mov3q_l,move_l,mvsz,
				  tst,clr,cmp,move")
		 (and (eq_attr "op_mem" "00")
		      (eq_attr "m68060_pair" "first")))))
  "m68060_pOEP")

(define_insn_reservation "m68060_first_mem" 1
  (and (eq_attr "cpu" "m68060")

;; Alone with a memory access, and read-modify-write (op_mem "11", "i1"
;; and "1i").
(define_insn_reservation "m68060_alone_mem" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "alu_l,aluq_l,cmp_l,clr_l,tst_l,neg_l,ext,
				  shift,moveq_l,mov3q_l,move_l,mvsz,
				  tst,clr,cmp,move")
		 (ior (eq_attr "op_mem" "11,i1,1i")
		      (and (eq_attr "op_mem" "10,i0,01,0i")
			   (eq_attr "m68060_pair" "alone"))))))
  "m68060_single + m68060_mem")

;; --- lea: an address computation, no memory access ------------------------

(define_insn_reservation "m68060_lea" 1

;; --- pOEP-but-allows-sOEP: Scc and Bcc ------------------------------------
;; These go in pOEP but don't block sOEP from dispatching.  An Scc to memory
;; takes the memory unit, so its sOEP partner can't access memory (Test 4).
;; Bcc is pOEP-only, but pOEP-but-allows-sOEP when the branch cache didn't
;; predict it and it branches forward, or predicted it not taken (Table
;; 10-2, note 1): the model assumes the latter.
(define_insn_reservation "m68060_scc" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "scc")
		 (eq_attr "op_mem" "00"))))
  "m68060_pOEP")

(define_insn_reservation "m68060_scc_mem" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (and (eq_attr "type" "scc")
		 (eq_attr "op_mem" "!00"))))
  "m68060_pOEP + m68060_mem")

(define_insn_reservation "m68060_bcc" 1
  (and (eq_attr "cpu" "m68060")
       (and (eq_attr "sched_insns" "1")
	    (eq_attr "type" "bcc")))
  "m68060_pOEP")
